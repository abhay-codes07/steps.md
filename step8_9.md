```python
import importlib.util
import subprocess
import sys

from agentmd2cpn.dsl import AgentSpec
from agentmd2cpn.emit import emit_dot, emit_json, emit_python, run_generated, write_spec
from agentmd2cpn.verify import build

SPEC = AgentSpec.model_validate({
    "name": "demo", "source_file": "demo/.Agent.md",
    "data": {"BRANCH": {"type": "enum", "values": ["main", "other"]}},
    "actions": [{"id": "run_tests"}, {"id": "git_commit"}, {"id": "git_push", "params": {"branch": "BRANCH"}},
                {"id": "write_secret"}],
    "rules": [
        {"id": "R1", "kind": "precedence", "args": {"first": "run_tests", "then": "git_commit"},
         "source": {"line_begin": 9, "quote": "Always run the test suite before committing."}},
        {"id": "R2", "kind": "limit", "args": {"action": "run_tests", "n": 3}},
        {"id": "R3", "kind": "forbid_when", "args": {"action": "git_push", "where": "branch == 'main'"}},
        {"id": "R4", "kind": "forbid", "args": {"action": "write_secret"}},
    ]})


def test_generated_file_is_standalone(tmp_path):
    frag, cpn, m0, ctx = build(SPEC)
    py = emit_python(SPEC, frag, cpn, m0, tmp_path / "demo_cpn.py")
    text = py.read_text(encoding="utf-8")
    assert "Always run the test suite before committing." in text and "'R1': ('precedence'" in text
    r = subprocess.run([sys.executable, str(py)], capture_output=True, text=True, timeout=300)
    assert r.returncode == 0, r.stderr
    assert "reachable markings: 8" in r.stdout and "DONE reachable (rules consistent): True" in r.stdout
    assert "trap states: 0" in r.stdout and "dead transitions: ['write_secret']" in r.stdout
    assert run_generated(py)
    s = importlib.util.spec_from_file_location("demo_cpn", py)
    mod = importlib.util.module_from_spec(s)
    s.loader.exec_module(mod)
    mon = mod.Monitor()
    assert not mon.try_action("git_commit") and mon.try_action("run_tests") and mon.try_action("git_commit")
    assert not mon.try_action("git_push", branch="main") and mon.can_finish()


def test_json_yaml_png(tmp_path):
    frag, cpn, m0, ctx = build(SPEC)
    j = emit_json(cpn, m0, ctx, tmp_path / "demo_cpn.json")
    assert j.exists() and "S_R1" in j.read_text(encoding="utf-8")
    y = write_spec(SPEC, tmp_path / "agent_spec.yaml")
    assert "kind: precedence" in y.read_text(encoding="utf-8")
    png = emit_dot(cpn, m0, tmp_path / "demo_cpn.png")
    assert png is None or png.exists()                        # None only when graphviz 'dot' is not installed
```

# Step 9 — extract.py: .Agent.md → DSL with an LLM (grounded + voted)

The LLM only fills the JSON schema of the DSL; it never writes Petri-net code. Every rule must
carry a verbatim quote from the file; a rule whose quote is not in the cited lines is rejected
(hallucination guard) and a rule seen in fewer than `min_votes` samples is dropped. The test uses a
fake LLM, so no API is needed yet — the Azure OpenAI wrapper is Step 10.

## 9.1 Test — `tests/test_extract.py` (39 lines)

```python
import json
from pathlib import Path

from agentmd2cpn.extract import LLMExtractor, extract_spec
from agentmd2cpn.ingest import parse_markdown

FIX = Path(__file__).parent / "fixtures" / "a" / ".Agent.md"
ACTS = [{"id": "run_tests"}, {"id": "git_commit"}, {"id": "git_push"}]
GOOD = {"kind": "precedence", "args": {"first": "run_tests", "then": "git_commit"},
        "quote": "Run pnpm test before committing.", "line_begin": 12, "line_end": 12}
HALLU = {"kind": "forbid", "args": {"action": "git_push"},
         "quote": "Never push to main.", "line_begin": 11, "line_end": 11}
MINOR = {"kind": "forbid", "args": {"action": "git_push"},
         "quote": "Never push directly to main.", "line_begin": 11, "line_end": 11}


def fake_llm(samples):
    it = iter(samples)
    return lambda prompt: next(it)


def test_prompt_shows_segments_and_kinds():
    doc = parse_markdown(FIX)
    p = LLMExtractor(lambda _: "{}").prompt(doc)
    assert "[s8 L12-12 list_item] Run pnpm test before committing." in p
    assert "precedence" in p and "phase_sequence" in p and '"quote"' in p


def test_grounding_and_voting():
    doc = parse_markdown(FIX)
    samples = [json.dumps({"actions": ACTS, "rules": [GOOD, HALLU]}),
               json.dumps({"actions": ACTS, "rules": [GOOD, MINOR]}),
               "not json at all",
               json.dumps({"actions": ACTS, "rules": [GOOD]})]
    spec = extract_spec(doc, "demo", LLMExtractor(fake_llm(samples), samples=4), min_votes=2)
    assert [r.kind for r in spec.rules] == ["precedence"]
    r = spec.rules[0]
    assert r.id == "R1" and r.votes == 3 and r.source.line_begin == 12 and r.source.quote == GOOD["quote"]
    assert [a.id for a in spec.actions] == ["run_tests", "git_commit", "git_push"]
```
